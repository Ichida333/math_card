\section{自然数と可算集合}
\label{section:自然数と可算集合}

\begin{definition}[p16]
	\label{definition:集合論的な自然数の定義}
	以下のように,自然数$n$を集合論的に構成できる.\\
	$n = n - 1 \cup \{n - 1\}$と定義する.すると,
	$n = \{0,1,2,... n-1\}と表せる.$
\end{definition}

\begin{definition}[p183の定義7.1.1]
	\label{definition:有限集合,無限集合の定義}
	$X$を集合とする。$X$が有限集合,あるいは無限集合であることを以下のように定義する.\\
	 $X$は有限集合$\defarr \exists n \in \nat\exists f( \; \BijectionMap{f}{n}{X} \; )$\\
	 $X$は無限集合$\defarr X$が無限集合でない.\\
\end{definition}

\begin{proposition}[p184の命題7.1.2]
	1. $n,m$を自然数とし,$f:n \rightarrow m$を全単射とする.このとき,
	$n=m$である.論理式は以下.\\
	$\forall n ,m \in \nat (\exists f(\BijectionMap{f}{n}{m}) \rightarrow n = m)$\\
	2. 自然数全体の集合$\nat$は無限集合である.論理式は以下.\\
	$\forall n \in \nat \forall f( \lnot \; \BijectionMap{f}{n}{\nat} \; )$
\end{proposition}

\begin{definition}[p184の定義7.1.3]
	\label{definition:合論的な自然数の定義}
	$X$を集合とする.\\
	1. $n$を自然数とする.
	全単射$n \rightarrow X$が存在するとき,
	$n$を$X$の濃度と呼び,
	$n=\Card(X)$と表す.\\
	2. 全単射$\nat \rightarrow X$が存在するとき,
	$X$は可算無限集合であるといい,
	$\Card(X)=\aleph_0$と表す.\\
	$X$が有限集合であるか無限集合であるとき,
	$X$は可算であるという.
\end{definition}

\begin{proposition}[p185の命題7.1.5]
	部分集合$A \subset \nat$に対し,
	関数$\map{C_A}{\nat}{\nat}$を帰納的に$C_A(0)=0$と,
  \begin{equation*}
    C_A(m + 1)=
    \begin{cases}
      C_A(m) + 1 & m \in A\text{のとき},\\
      C_A(m)  & m \not\in A \text{のとき}
    \end{cases}
  \end{equation*}
  で定める.\\
  1. $n$を自然数とし,
  $A \subseteq n$とする.
  このとき,
  $\Card(A)=C_A(n) \leq n$であり,
  $C_A$の制限は全単射$\map{\seigen {C_A}{A}}{A}{\Card(A)}$を定める.
  また,
  $C_A(n)=n$ならば,
  $A=n$である.論理式は以下.\\
 $\forall n \in \nat(A \subseteq n \rightarrow ((\Card(A)=C_A(n) \leq n ) \land (\BijectionMap{C_A\restriction_{{A}}}{A}{\Card(A)}
 ) \land (C_A(n) = n \rightarrow A=n)))$\\
 2. $A \subseteq n$をみたす自然数$n$は存在しないと仮定する.このとき,
 $\map {\seigen {C_A}{A}}{A}{\nat}$は全単射である.論理式は以下.\\
 $\lnot \exists n (A \subseteq n) \rightarrow \BijectionMap{\seigen{C_A}{A}}{A}{\nat}$
 
\end{proposition}
