  $\mathbf{claim1}$\\
  $n$は自然数とし,
  $A \subseteq n$とする.\\
  $v_n$を$v_n(A,0) = A$,
  $v_n(A,1) = A \cup \{n\}$とおくと,
  $\BijMap{v_n}{P(n) \times 2}{P(n+1)}$\\
  $\mathbf{proof}$\\
  $P(n+1) = \{X | \exists A \in P(n)( X = A \lor X = A \cup \{n\})\}$をまず示す.
  \begin{itemize}
    \item[]\hspace{-0.5cm} ($P(n+1) \subseteq *$) \\
      任意に$X \in P(n+1)$をとる.
      このとき$X \subseteq n+1$.
      よって$X \in P(n)$.
      \begin{itemize}
        \item[]\hspace{-0.5cm} ($n \not \in X$のとき) \\
          $A:= X$とおく.
          このとき,
          $X = A \land X = A \in P(n)$が真.\\
          そんな$A$の存在より$X \in *.$
        \item[]\hspace{-0.5cm} ($n \in X$のとき) \\
          $A:= X \setminus \{n\}$とおくと,
          $X = A \cup \{n\}$.\\
          また,
          $A = X \setminus \{n\} \in P(n)$.\\
          そんな$A$の存在より$X \in *.$
      \end{itemize}
    \item[]\hspace{-0.5cm} ($* \subseteq P(n+1)$) \\
      任意に$X \in *$をとると,
      $\exists A \in P(n)( X = A \lor X = A \cup \{n\})$が真より,
      そんな$A$を一つとり固定.
      \begin{itemize}
        \item[]\hspace{-0.5cm} ($X = A$のとき) \\
          $X = A \subseteq n$より,
          $X \in P(n+1)$.
        \item[]\hspace{-0.5cm} ($X = A  \cup \{n\}$のとき) \\
          $X = A  \cup \{n\} \subseteq n+1$より,
          $X \in P(n+1)$.
      \end{itemize}
  \end{itemize}
  よって$P(n) = \{A\},P(n+1) = \{A, A \cup \{n\}\}$と表せるので,
  $\BijMap{v_n}{P(n) \times 2}{P(n+1)}$.\\
  $\mathbf{claim2}$\\
  $n$は自然数とし,
  $m \in 2^n$とする.
  $u_n$を$u_n(m,0) = m$,
  $u_n(m,1) = m+ 2^n$と定める.\\
  このとき$\BijMap{u_n}{2^n \times 2}{2^{n+1}}$.\\
  $\mathbf{proof}$\\
  $2^{n+1} = \{x | \exists m (0 \leq m < 2^n \land( x = m \lor  x = m+ 2^n))\}$をまず示す.
  \begin{itemize}
    \item[]\hspace{-0.5cm} ($2^{n+1} \subseteq *$) \\
      任意に$x \in 2^{n+1}$をとると,
      $0 \leq x < 2^{n+1}$.
      \begin{itemize}
        \item[]\hspace{-0.5cm} ($0 \leq x < 2^n$のとき) \\
          $m:= x$とおくと,
          $m = 0 \leq x < 2^n$.\\
          よって$x =m \land 0 \leq m < 2^n$が真.\\
          そんな$m$の存在より$x \in *$.
        \item[]\hspace{-0.5cm} ($2^n \leq x < 2^{n+1}$のとき) \\
          $m:= x - 2^n$とおくと,
          $0 \leq x - 2^n < 2^{n+1} - 2^n = 2^n$より,
          $0 \leq m < 2^n$\\
          $x = m + 2^n$より,
          $0 \leq m < 2^n \land x = m+2^n$が真.\\
          そんな$m$の存在より$m \in *$.
      \end{itemize}
    \item[]\hspace{-0.5cm} ($ * \subseteq2^{n+1}$) \\
      任意に$x \in *$をとると,
      $\exists m (m < 2^n \land( x = m \lor  x = m+ 2^n))$が真より,
      そんな$m$を一つとり固定.
      \begin{itemize}
        \item[]\hspace{-0.5cm} ($x = m$のとき) \\
          $x = m < 2^n < 2^{n+1}$より,
          $x \in 2^{n+1}$
        \item[]\hspace{-0.5cm} ($x = m + 2^n$のとき) \\
          $x = m+ 2^n < 2^n + 2^n = 2^{n+1}$より,
          $x \in 2^{n+1}$
      \end{itemize}
  \end{itemize}
  よって$2^n = \{m\}, 2^{n+1} = \{m, m+2^n\}$と表せるので,
  $\BijMap{u_n}{2^n \times 2}{2^{n+1}}$.\\
$\mathbf{claim3}$\\
  $\nat = \bigcup _{n \in \nat} 2^n$\\
  $\mathbf{proof}$
    \begin{itemize}
      \item[]\hspace{-0.5cm} ((1) $\subseteq$ (2))\\
        任意に$x \in \nat$をとる.\\
        $n := x$とおくと,
        $x \in 2^n$.
        \begin{sub-block}{{\boldmath $\because$}\,}
          \begin{itemize}
            \item[] \hspace{-0.5cm} \\
              $\forall n \in \nat (n \in 2^n)$を帰納法で示す.
                \begin{itemize}
                  \item[]\hspace{-0.5cm} (base step)\\
                    $0 < 2^0 = 1$.
                  \item[]\hspace{-0.5cm} (induction step)\\
                    帰納法の仮定より$n < 2^n$.\\
                    $n+1 < 2^n + 1< 2^n + 2^n = 2^{n+1}$
                \end{itemize}
          \end{itemize}
        \end{sub-block}
        $\bigcup _{n \in \nat} 2^n \Leftrightarrow \{x | \exists n \in \nat (x \in 2^n)\}$と,
        $n \in \nat \land x \in 2^n$より
        そんな$n$の存在より,
        $x \in \bigcup _{n \in \nat} 2^n$.
      \item[]\hspace{-0.5cm} ((2) $\subseteq$ (1))\\
        任意に$x \in \bigcup _{n \in \nat} 2^n$をとると,
        $\exists n \in \nat (x \in 2^n)$が真より,
        そんな$n$を一つとり固定.\\
        $2^n \subseteq \nat$と$x \in 2^n$より,
        $x \in \nat$.
    \end{itemize}
\begin{proof} 1. \\
  $\forall n \in \nat (\BijMap{b_n}{P(n)}{2^n})$を数学的帰納法で示す.
  \begin{itemize}
    \item[]\hspace{-0.5cm} (base step)goal $:\Leftrightarrow \BijMap{b_0}{P(0)}{1}$\\
      $P(0) = \{\emptyset\}$,
      $1 = \{0\}$と$b_0(\emptyset) = 0$より,
      $b_0$は全単射.
    \item[]\hspace{-0.5cm} (induction step)\\
      $b_{n+1} \after u_n = v_n \after (b_n \times id_2)$を示す.\\
      $P(n) \times 2 = \{<A, 0>, <A,1>\}$より,$<A,0>,<A,1>$のときで場合分けする.
      \begin{itemize}
        \item[]\hspace{-0.5cm} ($<A, 0>$)\\
          $v_n(<A,0>) = A.$\\
          $b_{n+1}(A) = \sum _{m \in A} 2^m$.
          よって$b_{n+1} \after u_n(<A,0>) = \sum _{m \in A} 2^m$\\
          $(b_n \times id_2)(<A,0>) = <b_n(A), 0>$.\\
          $u_n(<b_n(A), 0>) = b_n(A) = \sum _{m \in A} 2^m$.\\
          よって$b_{n+1} \after u_n(<A,0>) = v_n \after (b_n \times id_2)(<A,0>)$.
        \item[]\hspace{-0.5cm} ($<A, 1>$)\\
          $v_n(<A,1>) = A \cup \{n\}.$\\
          $b_{n+1}(A \cup \{n\}) = \sum _{m \in A \cup \{n\}} 2^m = \sum _{m \in A} 2^m + 2^n$.\\
          よって$b_{n+1} \after u_n(<A,1>) = \sum _{m \in A} 2^m + 2^n$\\
          $(b_n \times id_2)(<A,1>) = <b_n(A), 1>$.\\
          $u_n(<b_n(A), 1>) = b_n(A) + 2^n = \sum _{m \in A} 2^m + 2^n$.\\
          よって$b_{n+1} \after u_n(<A,1>) = v_n \after (b_n \times id_2)(<A,1>)$.
      \end{itemize}
      以上より,
      $b_{n+1} \after u_n = v_n \after (b_n \times id_2)$は真.\\
      今,
      $u_n, v_n, b_n \times id_2$は全単射なので,
      $b_{n+1}$も全単射になる.
  \end{itemize}
\end{proof} 
  \begin{proof} 2\\
  $\mathbf{claim4}$を適用すると,
  goal$\Leftrightarrow \BijMap{b}{F(\nat)}{\bigcup _{n \in \nat} 2^n}$.\\
  単射性と全射性に分けて証明する.
    \begin{itemize}
      \item[]\hspace{-0.5cm} (単射性)\\
        任意に$A,B \in F(\nat)$をとり,
        $b(A) = b(B)$とする.\\
        このとき$A,B$は有限集合かつ$A,B \subseteq \nat$なので,
        $\mathbf{Corollary7.1.6(1)}$より,
        $A \subseteq n, B \subseteq m$を満たす自然数$n,m$が存在するのでそれらを固定.\\
        $l := \operatorname{max}(n,m)$とおく.\\
        このとき$A,B \in P(l)$.
        また,
        $b_l(A) = b(A), b_l(B) = b(B)$より,
        $b_l(A) = b_l(b)$.\\
        1.より$b_l$は単射なので,
        $A = B$.
        よって$b$は単射.
      \item[]\hspace{-0.5cm} (全射性)\\
        任意に$x \in \bigcup _{n \in \nat} 2^n$をとる.
        このとき,
        $\exists n \in \nat(x \in 2^n)$が真より,
        そんな$n$を一つとり固定.\\
        また,
        1.より$b_n$は全射なので,
        $x \in 2^n$を代入すると,
        $\exists A \in P(n)(b_n(A) = x)$が真\\
        そんな$A$を一つとり固定.\\
        $A \subseteq n \subseteq \nat$.
        よって$A \in F(\nat)$.\\
        また,
        $b(A) = b_n(A) = x$.\\ 
        そんな$A$の存在より,
        $\exists A \in F(\nat) (b(A) = x)$は真.
        よって$b$は全射.
    \end{itemize}
\end{proof}
