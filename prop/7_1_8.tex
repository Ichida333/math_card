\begin{proposition}[p188の命題7.1.8]
  集合$X$に対し,
  $F(X) = \{A \in P(X) | Aは有限集合\}$とする.\\
  1. $A \subseteq n$を$b(A) = \sum _{m \in A} 2^m \in 2^n$にうつす写像を$b_n$とすると,
  $\BijMap{b_n}{P(n)}{2^n}$.\\
  2. 有限部分集合$A \subseteq \nat$を$b(A) = \sum _{m \in A} 2^m$へうつす写像を$b$とすると,
  $\BijMap{b}{F(\nat)}{\nat}$.\\
\end{proposition}
