\begin{corollary}[p187の系7.1.7]
  $f$が$X$から$Y$の写像とする.\\
  1. $f$が単射とする.
  $Y$が可算集合のとき$X$も可算集合である.\\
  また,
  $Y$が有限集合のとき$X$も有限集合であり,
  $\Card{X} \leq \Card{Y}$である.\\
  さらにこのとき$\Card{X} = \Card{Y}$ならば$f$は全単射である.\\
  論理式は以下.\\
  \begin{equation*}
    fが単射 \rightarrow
    \begin{cases}
     Yが可算集合 \rightarrow Xが可算集合(1.1)\\
     Yが有限集合 \rightarrow Xが有限集合 \land \Card{X} \leq \Card{Y}(1.2)\\
     \Card{X} = \Card{Y} \rightarrow fは全単射(1.3)\\
    \end{cases}
  \end{equation*}
  2. $f$が全射とする.
  $Y$が可算集合なら$X$も可算集合である.\\
  $Y$が有限集合なら$X$も有限集合であり,
  $\Card{X} \geq \Card{Y}$である.\\
  さらにこのとき$\Card{X} = \Card{Y}$ならば$f$は全単射である.
  \begin{equation*}
    fが全射 \rightarrow
    \begin{cases}
     Xが可算集合 \rightarrow Yが可算集合(1.1)\\
     Xが有限集合 \rightarrow Yが有限集合 \land \Card{Y} \leq \Card{X}(1.2)\\
     \Card{X} = \Card{Y} \rightarrow fは全単射(1.3)\\
    \end{cases}
  \end{equation*}
\end{corollary}
