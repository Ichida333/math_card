\begin{proposition}[p185の命題7.1.5]
	部分集合$A \subset \nat$に対し,
	関数$\map{C_A}{\nat}{\nat}$を帰納的に$C_A(0)=0$と,
  \begin{equation*}
    C_A(m + 1)=
    \begin{cases}
      C_A(m) + 1 & m \in A\text{のとき},\\
      C_A(m)  & m \not\in A \text{のとき}
    \end{cases}
  \end{equation*}
  で定める.\\
  1. $n$を自然数とし,
  $A \subseteq n$とする.
  このとき,
  $\Card(A)=C_A(n) \leq n$であり,
  $C_A$の制限は全単射$\map{\seigen {C_A}{A}}{A}{\Card(A)}$を定める.
  また,
  $C_A(n)=n$ならば,
  $A=n$である.論理式は以下.
  
 \begin{equation*}
   \forall n \in \nat(A \subseteq n \rightarrow ((\Card{A}=C_A(n) \leq n ) \land (\BijectionMap{C_A\restriction_{{A}}}{A}{\Card{A}}
 ) \land (C_A(n) = n \rightarrow A=n)))
 \end{equation*}
 2. $A \subseteq n$をみたす自然数$n$は存在しないと仮定する.このとき,
 $\map {\seigen {C_A}{A}}{A}{\nat}$は全単射である.論理式は以下.
 \begin{equation*}
 \lnot \exists n (A \subseteq n) \rightarrow \BijectionMap{\seigen{C_A}{A}}{A}{\nat}
 \end{equation*}
\end{proposition}
